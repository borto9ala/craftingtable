Crafting Table

Rust terminal application

Current version: Crafting Table is now a Rust project. From the project directory, run cargo run; see README.md for current setup and backend requirements. The steps below document the legacy Bash version.

1 Install Homebrew

If the MacBook doesn't have Homebrew installed yet, open the terminal and run:

/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"

2 Install Dependencies

Homebrew has all required tools readily available in its official repositories:

brew install figlet imagemagick ffmpeg yt-dlp gallery-dl

3 Create the Script File

Create the scripts directory and open the file in nano:

mkdir -p ~/.scripts
nano ~/.scripts/craftingtable.sh

Paste the following code into the file, then save (Ctrl+O, Enter) and exit (Ctrl+X):

#!/bin/bash

# Print the aesthetic header
if command -v figlet &> /dev/null; then
    figlet "craftingtable"
else
    echo "======================================"
    echo "           CRAFTING TABLE             "
    echo "======================================"
fi

# Function to parse destination shortcuts
get_dest() {
    local input="$1"
    case "$input" in
        dz|"") echo "$HOME/Downloads" ;;
        dc) echo "$HOME/Documents" ;;
        img) echo "$HOME/Pictures" ;;
        mv) echo "$HOME/Movies" ;;
        ms) echo "$HOME/Music" ;;
        *) echo "$input" ;;
    esac
}

echo "What would you like to craft today?"
echo "1) 🔄 Format Conversion (Media/Images)"
echo "2) 📐 Resolution Scaling (Video)"
echo "3) 📥 Mass YouTube Download (yt-dlp)"
echo "4) 🖼️ Mass Image Conversion (magick)"
echo "5) 🌐 Mass Image Download (gallery-dl)"
read -p "Select an option (1-5): " opt

case $opt in
    1)
        read -p "File to convert (drag & drop or path): " src_file
        read -p "Target format (e.g., mp4, mp3, png, jpg): " ext
        read -p "Destination (dz, dc, img, mv, ms, or path) [Default: dz]: " dest_input
        
        dest_dir=$(get_dest "$dest_input")
        mkdir -p "$dest_dir"
        
        filename=$(basename -- "$src_file")
        base="${filename%.*}"
        
        echo "Converting..."
        ffmpeg -i "$src_file" "$dest_dir/$base.$ext"
        echo "✅ Done! Saved to $dest_dir/$base.$ext"
        ;;

    2)
        read -p "Video file to scale: " src_file
        read -p "Target height (e.g., 1080 for FHD, 1440 for 2K, 2160 for 4K): " res
        read -p "Orientation? (v for vertical, h for horizontal): " orientation
        read -p "Destination (dz, dc, img, mv, ms, or path) [Default: dz]: " dest_input
        
        dest_dir=$(get_dest "$dest_input")
        mkdir -p "$dest_dir"
        filename=$(basename -- "$src_file")
        base="${filename%.*}"
        
        if [ "$orientation" == "v" ]; then
            scale_filter="scale=-2:$res"
        else
            scale_filter="scale=-2:$res"
        fi
        
        echo "Scaling..."
        ffmpeg -i "$src_file" -vf "$scale_filter" -vcodec libx264 -crf 23 -acodec aac "$dest_dir/${base}_${res}p.mp4"
        echo "✅ Done! Saved to $dest_dir/${base}_${res}p.mp4"
        ;;

    3)
        read -p "Paste YouTube links (separated by space): " links
        read -p "Format (mp4, mp3) [Default: mp4]: " format
        read -p "Destination (dz, dc, img, mv, ms, or path) [Default: dz]: " dest_input
        
        dest_dir=$(get_dest "$dest_input")
        mkdir -p "$dest_dir"
        
        if [ "$format" == "mp3" ]; then
            yt-dlp -x --audio-format mp3 -P "$dest_dir" $links
        else
            yt-dlp -S res,ext:mp4:m4a --recode mp4 -P "$dest_dir" $links
        fi
        echo "✅ Mass download complete!"
        ;;

    4)
        read -p "Source folder (path): " src_dir
        read -p "Source file extension (e.g., webp, jpg): " src_ext
        read -p "Target format (e.g., png): " target_ext
        read -p "Destination (dz, dc, img, mv, ms, or path) [Default: dz]: " dest_input
        
        dest_dir=$(get_dest "$dest_input")
        mkdir -p "$dest_dir"
        
        echo "Mass converting images..."
        magick "$src_dir/*.$src_ext" "$dest_dir/%d.$target_ext"
        echo "✅ Done! Images saved in $dest_dir"
        ;;

    5)
        read -p "Website link to download from: " link
        read -p "Destination (dz, dc, img, mv, ms, or path) [Default: dz]: " dest_input
        
        dest_dir=$(get_dest "$dest_input")
        mkdir -p "$dest_dir"
        
        echo "Fetching gallery..."
        gallery-dl -d "$dest_dir" "$link"
        echo "✅ Done! Gallery downloaded to $dest_dir"
        ;;

    *)
        echo "Invalid option. Exiting."
        ;;
esac

4 Make the Script Executable

Grant execution permissions so the script can run:

chmod +x ~/.scripts/craftingtable.sh

5 Add the Alias to Zsh

Open your Zsh configuration file:

nano ~/.zshrc

Add the following alias at the bottom of the file:

alias craftingtable="~/.scripts/craftingtable.sh"

6 Apply the Changes

Refresh your terminal session to activate the alias:

source ~/.zshrc
